{{/* Override of PaperModX's image render hook. Upstream only resolved page resources and assets and warned for anything else. This version: - resolves assets whether or not the path has a leading slash - resizes raster images found in assets/ to the content column width and emits a width-based srcset, so posts don't ship full-resolution files. Format is kept as-is: a `webp q82` in the Resize spec would shrink these further, at the cost of handing visitors .webp on save. - falls back to serving the path as-is for files that live in static/ */}} {{- $url := urls.Parse .Destination -}} {{- if $url.Scheme -}} {{- else -}} {{/* page resource, then assets with and without the leading slash */}} {{- $img := .Page.Resources.GetMatch .Destination -}} {{- if not $img }}{{ $img = resources.Get (strings.TrimPrefix "/" .Destination) }}{{ end -}} {{- if not $img }}{{ $img = resources.Get .Destination }}{{ end -}} {{- $processable := slice "png" "jpeg" "webp" -}} {{- $contentWidth := 720 -}} {{- if and $img (eq $img.MediaType.MainType "video") -}} {{- else if and $img (in $processable $img.MediaType.SubType) -}} {{/* candidate widths, never upscaling past the source */}} {{- $widths := slice -}} {{- range (slice 480 $contentWidth (mul $contentWidth 2)) -}} {{- if le . $img.Width }}{{ $widths = $widths | append . }}{{ end -}} {{- end -}} {{- if not (in $widths $img.Width) }}{{ $widths = $widths | append $img.Width }}{{ end -}} {{- $srcset := slice -}} {{- range $widths -}} {{- $r := $img.Resize (printf "%dx" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink .) -}} {{- end -}} {{/* default src: the largest width at or below the content column */}} {{- $defaultWidth := $img.Width -}} {{- range $widths }}{{ if le . $contentWidth }}{{ $defaultWidth = . }}{{ end }}{{ end -}} {{- $default := $img.Resize (printf "%dx" $defaultWidth) -}} {{- else -}} {{/* svg/gif in assets, or anything served straight from static/ */}} {{- $src := .Destination -}} {{- with $img }}{{ $src = .RelPermalink }}{{ end -}} {{- end -}} {{- end -}}
