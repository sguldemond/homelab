{{/* Override of PaperModX's image render hook: upstream only resolves page resources and assets, and warns for anything else. We fall back to serving the path as-is so images living in static/ just work. */}} {{- $url := urls.Parse .Destination -}} {{- if $url.Scheme -}} {{- else -}} {{- $img := .Page.Resources.GetMatch .Destination -}} {{- if not $img -}} {{- $img = resources.Get .Destination -}} {{- end -}} {{- if and $img (eq $img.MediaType.MainType "video") -}} {{- else -}} {{- $src := .Destination -}} {{- with $img }}{{ $src = .RelPermalink }}{{ end -}} {{- end -}} {{- end -}}
