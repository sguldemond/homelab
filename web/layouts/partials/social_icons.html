{{/* Override of PaperModX's social_icons: upstream renders the bare SVG with the name only as a hover tooltip. This adds an optional visible label, set per icon with `label:` in socialIcons (omit it to get icon-only behaviour back). */}}
{{- range .icons }} {{- $icon_name := ( trim .name " " | lower )}} {{- $href := trim .url " " }} {{- $is_email := eq .url "email" }} {{- if $is_email }} {{- $href = printf "mailto:%s" ($.MaskedEmail | default $.Email) }} {{- end }} {{ safeHTML (index $.svg $icon_name | default $.svg.default) }} {{- with .label }}{{ . }}{{ end }} {{- end }}
