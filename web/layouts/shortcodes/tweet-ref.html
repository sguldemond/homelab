{{/* Override of PaperModX's shortcode: the upstream version uses `getJSON`, which was removed from Hugo in v0.157.0. Same behaviour via resources.GetRemote. */}}
{{- $url := .Get 0 -}} From {{ strings.TrimPrefix "https://" $url }}: {{- $query := querify "url" $url "dnt" true -}} {{- $request := printf "https://publish.twitter.com/oembed?%s" $query -}} {{- with try (resources.GetRemote $request) -}} {{- with .Err -}} {{- warnf "tweet-ref: %s" . -}} {{- else with .Value -}} {{- $json := unmarshal .Content -}} {{- replaceRE "" "" $json.html | safeHTML -}} {{- end -}} {{- end -}}
